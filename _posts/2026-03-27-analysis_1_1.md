    <h3 style="text-align: center;">ε-δ 시각화: f(x) = 0.1x³ - 1.1x² + 3x + 1</h3>
    
    <canvas id="ed-canvas" style="width: 100%; height: 400px; background: #fafafa; border: 1px solid #eee;"></canvas>
    
    <div style="margin-top: 20px; display: flex; flex-direction: column; gap: 10px;">
        <label><strong>ε (오차 허용 범위):</strong> <span id="epsilon-val">1.00</span></label>
        <input type="range" id="epsilon-slider" min="0.05" max="2.0" step="0.01" value="0.5" style="width: 100%;">
        
        <div style="background: #eef2ff; padding: 15px; border-radius: 8px; font-size: 0.9em; line-height: 1.6;">
            <strong>정의:</strong> 임의의 <span style="color: blue;">ε > 0</span>에 대하여, <br>
            0 < |x - a| < <span style="color: green;">δ</span> 이면 |f(x) - L| < <span style="color: blue;">ε</span> 를 만족하는 <span style="color: green;">δ > 0</span>가 존재하는가?

<script>
const canvas = document.getElementById('ed-canvas');
const ctx = canvas.getContext('2d');
const slider = document.getElementById('epsilon-slider');
const epsilonText = document.getElementById('epsilon-val');

// 설정값
const a = 4; // 목표 x점
const f = (x) => 0.1 * Math.pow(x, 3) - 1.1 * Math.pow(x, 2) + 3 * x + 1;
const L = f(a); // 목표 y점 (극한값)

// 좌표계 변환 함수
const padding = 50;
const mapX = (x) => padding + (x * (canvas.width - 2 * padding) / 8);
const mapY = (y) => (canvas.height - padding) - (y * (canvas.height - 2 * padding) / 10);

function draw() {
    canvas.width = canvas.offsetWidth * window.devicePixelRatio;
    canvas.height = canvas.offsetHeight * window.devicePixelRatio;
    ctx.scale(window.devicePixelRatio, window.devicePixelRatio);
    
    const eps = parseFloat(slider.value);
    epsilonText.innerText = eps.toFixed(2);

    ctx.clearRect(0, 0, canvas.width, canvas.height);

    // 1. 축 그리기
    ctx.strokeStyle = '#ccc';
    ctx.beginPath();
    ctx.moveTo(padding, 0); ctx.lineTo(padding, canvas.height - padding);
    ctx.lineTo(canvas.width, canvas.height - padding);
    ctx.stroke();

    // 2. ε (Epsilon) 영역 (가로 띠)
    ctx.fillStyle = 'rgba(59, 130, 246, 0.1)';
    ctx.fillRect(padding, mapY(L + eps), canvas.width - 2*padding, mapY(L - eps) - mapY(L + eps));
    
    // 3. 적절한 δ (Delta) 계산 (수치적 근사)
    let delta = 0;
    for (let d = 0.01; d < 2; d += 0.01) {
        let y1 = f(a + d);
        let y2 = f(a - d);
        if (Math.abs(y1 - L) < eps && Math.abs(y2 - L) < eps) {
            delta = d;
        } else {
            break;
        }
    }

    // 4. δ (Delta) 영역 (세로 띠)
    ctx.fillStyle = 'rgba(34, 197, 94, 0.2)';
    ctx.fillRect(mapX(a - delta), 0, mapX(a + delta) - mapX(a - delta), canvas.height - padding);

    // 5. 함수 그래프 그리기
    ctx.strokeStyle = '#333';
    ctx.lineWidth = 2;
    ctx.beginPath();
    for (let x = 0; x <= 8; x += 0.1) {
        let px = mapX(x);
        let py = mapY(f(x));
        if (x === 0) ctx.moveTo(px, py);
        else ctx.lineTo(px, py);
    }
    ctx.stroke();

    // 6. 목표점 (a, L) 표시
    ctx.fillStyle = 'red';
    ctx.beginPath();
    ctx.arc(mapX(a), mapY(L), 4, 0, Math.PI * 2);
    ctx.fill();

    // 가이드 라인 텍스트
    ctx.fillStyle = '#666';
    ctx.font = '12px Arial';
    ctx.fillText(`L+ε`, 10, mapY(L+eps));
    ctx.fillText(`L-ε`, 10, mapY(L-eps));
    ctx.fillText(`a`, mapX(a)-5, canvas.height-padding+20);

    requestAnimationFrame(draw);
}

slider.addEventListener('input', draw);
window.addEventListener('resize', draw);
draw();
</script>
