    <canvas id="canvas" style="background: white; border: 1px solid #ddd; cursor: crosshair;"></canvas>
    <div style="margin-top: 10px; font-family: sans-serif;">
        <strong>마우스를 움직여 각도를 조절하고, 클릭해서 발사하세요!</strong><br>
        <span id="info">각도: 0° | 파워: 0</span>
    </div>
</div>

<script>
const canvas = document.getElementById('canvas');
const ctx = canvas.getContext('2d');
const info = document.getElementById('info');

// 설정값
canvas.width = 700;
canvas.height = 400;
const gravity = 0.5; // 중력 가속도
const startX = 50;
const startY = canvas.height - 50;

let mouseX = 0, mouseY = 0;
let isFlying = false;
let ball = { x: startX, y: startY, vx: 0, vy: 0 };
let trajectory = [];

// 마우스 움직임 감지
canvas.addEventListener('mousemove', (e) => {
    const rect = canvas.getBoundingClientRect();
    mouseX = e.clientX - rect.left;
    mouseY = e.clientY - rect.top;
});

// 클릭 시 발사
canvas.addEventListener('mousedown', () => {
    if (isFlying) return; // 이미 날아가고 있으면 무시
    
    const dx = mouseX - startX;
    const dy = mouseY - startY;
    const angle = Math.atan2(-dy, dx);
    const power = Math.min(Math.sqrt(dx*dx + dy*dy) / 10, 20); // 거리에 비례한 파워

    ball.vx = Math.cos(angle) * power;
    ball.vy = -Math.sin(angle) * power;
    ball.x = startX;
    ball.y = startY;
    isFlying = true;
    trajectory = [];
});

function draw() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    // 1. 가이드 라인 (각도 조절선)
    if (!isFlying) {
        const dx = mouseX - startX;
        const dy = mouseY - startY;
        const angle = Math.atan2(-dy, dx);
        const power = Math.min(Math.sqrt(dx*dx + dy*dy) / 10, 20);
        
        ctx.beginPath();
        ctx.setLineDash([5, 5]);
        ctx.moveTo(startX, startY);
        ctx.lineTo(mouseX, mouseY);
        ctx.strokeStyle = '#aaa';
        ctx.stroke();
        ctx.setLineDash([]);
        
        info.innerText = `각도: ${Math.round(angle * 180 / Math.PI)}° | 파워: ${Math.round(power * 10)}`;
    }

    // 2. 궤적 그리기
    ctx.beginPath();
    ctx.strokeStyle = '#3498db';
    trajectory.forEach((p, i) => {
        if (i === 0) ctx.moveTo(p.x, p.y);
        else ctx.lineTo(p.x, p.y);
    });
    ctx.stroke();

    // 3. 공 그리기
    ctx.beginPath();
    ctx.arc(ball.x, ball.y, 8, 0, Math.PI * 2);
    ctx.fillStyle = '#e74c3c';
    ctx.fill();

    // 4. 물리 계산
    if (isFlying) {
        ball.x += ball.vx;
        ball.vy += gravity; // 중력 적용
        ball.y += ball.vy;
        trajectory.push({x: ball.x, y: ball.y});

        // 바닥이나 벽에 닿으면 정지
        if (ball.y > canvas.height - 10 || ball.x > canvas.width) {
            isFlying = false;
            setTimeout(() => { // 1초 뒤 초기화
                ball.x = startX;
                ball.y = startY;
                trajectory = [];
            }, 1000);
        }
    }

    requestAnimationFrame(draw);
}

draw();
</script>
