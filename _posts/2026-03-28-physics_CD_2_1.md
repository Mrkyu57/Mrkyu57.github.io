    <h3>수직 방향의 힘의 평형</h3>
    <p>바닥에 놓인 물체를 위로 끌어당길 때 ($F_{ext}$):</p>
    <p style="font-size: 20px; color: #8354ce;">$$F_N = mg - F_{ext, y} \quad (\text{if } F_N > 0)$$</p>
    <p style="font-size: 14px; color: #666;">* 외력이 중력보다 커지면 물체는 바닥에서 떨어지며 수직항력($F_N$)은 0이 됩니다.</p>
</div>

<script>
    const canvas = document.getElementById('verticalCanvas');
    const ctx = canvas.getContext('2d');
    const massSlider = document.getElementById('massSlider');
    const massValue = document.getElementById('massValue');
    const forceModeBtn = document.getElementById('forceModeBtn');
    const resetBtn = document.getElementById('resetBtn');

    let mass = 5;
    const groundY = 350;
    let box = { x: 420, y: 0, w: 60, h: 60 };
    let isForceMode = false;
    let showForceVectors = false;
    let isDragging = false;
    let dragOffset = { x: 0, y: 0 };
    let mousePos = { x: 0, y: 0 };

    function updateBoxSize() {
        box.w = 40 + mass * 4;
        box.h = 40 + mass * 4;
        // 리셋 시에만 바닥으로
        if (!isDragging && box.y + box.h > groundY) {
            box.y = groundY - box.h;
        }
    }

    updateBoxSize();

    massSlider.addEventListener('input', (e) => {
        mass = parseFloat(e.target.value);
        massValue.innerText = mass.toFixed(1) + 'kg';
        updateBoxSize();
    });

    forceModeBtn.addEventListener('click', () => {
        isForceMode = !isForceMode;
        forceModeBtn.innerText = `Force Analysis: ${isForceMode ? 'ON' : 'OFF'}`;
        forceModeBtn.style.background = isForceMode ? "#8354ce" : "#475569";
        if(!isForceMode) showForceVectors = false;
    });

    resetBtn.addEventListener('click', () => {
        box.x = 450 - box.w/2;
        box.y = groundY - box.h;
        mass = 5;
        massSlider.value = 5;
        massValue.innerText = '5.0kg';
        isForceMode = false;
        showForceVectors = false;
        forceModeBtn.style.background = "#475569";
        forceModeBtn.innerText = "Force Analysis: OFF";
        updateBoxSize();
    });

    canvas.addEventListener('mousedown', (e) => {
        const rect = canvas.getBoundingClientRect();
        const mx = e.clientX - rect.left;
        const my = e.clientY - rect.top;

        if (mx > box.x && mx < box.x + box.w && my > box.y && my < box.y + box.h) {
            if (isForceMode) {
                showForceVectors = true;
            }
            isDragging = true;
            dragOffset.x = mx - box.x;
            dragOffset.y = my - box.y;
        }
    });

    window.addEventListener('mousemove', (e) => {
        const rect = canvas.getBoundingClientRect();
        mousePos.x = e.clientX - rect.left;
        mousePos.y = e.clientY - rect.top;

        if (isDragging) {
            box.x = mousePos.x - dragOffset.x;
            box.y = mousePos.y - dragOffset.y;

            // 바닥 아래로 내려가지 못하게 제한
            if (box.y + box.h > groundY) {
                box.y = groundY - box.h;
            }
        }
    });

    window.addEventListener('mouseup', () => {
        isDragging = false;
        if (!isForceMode) showForceVectors = false;
    });

    function draw() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);

        // 바닥 그리기
        ctx.strokeStyle = "#334155";
        ctx.lineWidth = 4;
        ctx.beginPath();
        ctx.moveTo(0, groundY);
        ctx.lineTo(canvas.width, groundY);
        ctx.stroke();

        // 바닥 패턴
        ctx.strokeStyle = "#cbd5e1";
        ctx.lineWidth = 1;
        for (let i = 0; i < canvas.width; i += 20) {
            ctx.beginPath(); ctx.moveTo(i, groundY); ctx.lineTo(i - 10, groundY + 15); ctx.stroke();
        }

        // 물체 그리기
        ctx.fillStyle = isDragging ? "#64748b" : "#94a3b8";
        ctx.fillRect(box.x, box.y, box.w, box.h);
        ctx.strokeStyle = "#1e293b";
        ctx.lineWidth = 2;
        ctx.strokeRect(box.x, box.y, box.w, box.h);

        // 질량 텍스트
        ctx.fillStyle = "#fff";
        ctx.font = "bold 14px Arial";
        ctx.textAlign = "center";
        ctx.fillText(`${mass}kg`, box.x + box.w/2, box.y + box.h/2 + 5);

        // 힘 분석 로직
        if (showForceVectors) {
            const centerX = box.x + box.w / 2;
            const centerY = box.y + box.h / 2;
            const F_gravity = mass * 15; // 시각적 스케일링
            
            let F_ext_x = 0;
            let F_ext_y = 0;
            
            if (isDragging) {
                F_ext_x = mousePos.x - (box.x + dragOffset.x);
                F_ext_y = mousePos.y - (box.y + dragOffset.y);
            }

            // 1. 중력 (언제나 존재)
            drawVector(centerX, centerY, centerX, centerY + F_gravity, "#ef4444", "Fg");

            // 2. 외력 (잡아끄는 힘)
            if (isDragging) {
                drawVector(centerX, centerY, centerX + F_ext_x, centerY + F_ext_y, "#10b981", "Fext");
            }

            // 3. 수직항력 계산 (바닥에 닿아있을 때만)
            const isTouchingGround = (box.y + box.h >= groundY - 1);
            if (isTouchingGround) {
                // 외력의 수직 성분 (위쪽이 -, 따라서 빼줌)
                // 실제 물리: Fn = mg - F_ext_up (여기서 F_ext_y가 음수면 위로 끄는 것)
                let pullUpForce = isDragging ? -F_ext_y : 0;
                let Fn_value = Math.max(0, F_gravity - pullUpForce);
                
                if (Fn_value > 0) {
                    drawVector(centerX, box.y + box.h, centerX, box.y + box.h - Fn_value, "#3b82f6", "FN");
                }
            }
        }

        requestAnimationFrame(draw);
    }

    function drawVector(x1, y1, x2, y2, color, label) {
        const dx = x2 - x1;
        const dy = y2 - y1;
        const len = Math.sqrt(dx*dx + dy*dy);
        if (len < 5) return;

        const headlen = 10;
        const angle = Math.atan2(dy, dx);
        
        ctx.strokeStyle = color;
        ctx.fillStyle = color;
        ctx.lineWidth = 3;

        ctx.beginPath();
        ctx.moveTo(x1, y1);
        ctx.lineTo(x2, y2);
        ctx.stroke();

        ctx.beginPath();
        ctx.moveTo(x2, y2);
        ctx.lineTo(x2 - headlen * Math.cos(angle - Math.PI / 6), y2 - headlen * Math.sin(angle - Math.PI / 6));
        ctx.lineTo(x2 - headlen * Math.cos(angle + Math.PI / 6), y2 - headlen * Math.sin(angle + Math.PI / 6));
        ctx.fill();

        ctx.font = "bold 16px 'Times New Roman'";
        ctx.shadowBlur = 4;
        ctx.shadowColor = "rgba(0,0,0,0.2)";
        ctx.fillText(label, x2 + 10 * Math.cos(angle), y2 + 10 * Math.sin(angle));
        ctx.shadowBlur = 0;
    }

    draw();
</script>
