       &#9654;
    </button>
    <button id="reset" style="padding: 8px 16px; cursor: pointer; background: #000000; color: white; border: none; border-radius: 4px; font-size: 17px;">
       &#8635;
    </button>
</div>


<canvas id="motionCanvas" width="600" height="200" style="border:1px solid #ddd; background: #f9f9f9; margin: 0 auto;"></canvas>

<script>
  const canvas = document.getElementById('motionCanvas');
  const ctx = canvas.getContext('2d');
  const start = document.getElementById('start');
  const reset = document.getElementById('reset');

  const radius = 15;                                      // 물체의 반지름
  const origin_position = { x: 300, y: 100 };             // 원점의 위치
  const initial_position = { x: 300, y: 100 };            // 처음 물체의 위치
  const initial_velocity = { x: 0.5, y: -0.1 };           // 처음 물체의 속도

  let position = { x: 300, y: 100 };                      // 위치 벡터
  let velocity = { x: 0.5, y: -0.1 };                     // 속도 벡터
  let isMoving = false;

  start.addEventListener('click', () => {
    isMoving = !isMoving;
    if (isMoving) {
        start.innerHTML = '<span>&#10074;&#10074;</span>';
        start.style.background = "#ffc107"; 
        start.style.color = "#000";
    } else {
        start.innerHTML = '<span>&#9654;</span>';
        start.style.background = "#28a745"; // 다시 초록색으로 변경
        start.style.color = "#fff";
    }
});

// 초기화 버튼 클릭 시 버튼 모양도 '재생'으로 되돌리기
reset.addEventListener('click', () => {
    isMoving = false;
    position = { ...origin_position };
    velocity = { ...initial_velocity };
    
    // 버튼 초기화
    start.innerHTML = '<span>&#9654;</span> 재생';
    start.style.background = "#28a745";
    start.style.color = "#fff";
});

  reset.addEventListener('click', () => {   
        isMoving = false;              
        position = { ...initial_position };               // 위치 초기화
        velocity = { ...initial_velocity };               // 속도 초기화
  });


  function draw() {   // 그리기 함수

    ctx.clearRect(0, 0, canvas.width, canvas.height);     //캔버스 초기화
    if (isMoving) {
        position.x += velocity.x;
        position.y += velocity.y;
    }
    //position.x += velocity.x;                             // x 위치 계산
    //position.y += velocity.y;                             // y 위치 계산
    const dispX = position.x - origin_position.x;         // 원점으로 부터의 x방향 위치 차이
    const dispY = position.y - origin_position.y;         // 원점으로 부터의 y방향 위치 차이

    drawMoveVector(origin_position.x, origin_position.y, dispX, dispY, "gray", "r");      // 변위벡터 그리기
    drawMoveVector(position.x, position.y, velocity.x * 80, velocity.y * 80, "red","v");      // 속도벡터 그리기
    
    ctx.fillStyle = "gray";
    ctx.beginPath();
    ctx.arc(origin_position.x, origin_position.y, 3, 0, Math.PI * 2);
    ctx.fill();

    ctx.beginPath();
    ctx.arc(position.x, position.y, 10, 0, Math.PI * 2);
    ctx.fillStyle = "#007bff";
    ctx.fill();
    ctx.closePath();

    requestAnimationFrame(draw);
  }


  function drawMoveVector(x, y, v_x, v_y, color, label) {
    if (v_x === 0 && v_y === 0) return;
    
    const headLength = 10;                
    const angle = Math.atan2(v_y, v_x);

    ctx.strokeStyle = color;
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(x, y);
    ctx.lineTo(x + v_x, y + v_y);
    ctx.stroke();

    ctx.beginPath();
    ctx.moveTo(x + v_x, y + v_y); // 화살표 끝점
    ctx.lineTo(
        x + v_x - headLength * Math.cos(angle - Math.PI / 7), 
        y + v_y - headLength * Math.sin(angle - Math.PI / 7)
    );
    ctx.lineTo(
        x + v_x - headLength * Math.cos(angle + Math.PI / 7), 
        y + v_y - headLength * Math.sin(angle + Math.PI / 7)
    );

    ctx.closePath();
    ctx.fillStyle = color;
    ctx.fill();
    ctx.font = "bold 14px Arial";
    ctx.fillStyle = color;
    ctx.textAlign = "center";
    
    const midX = x + v_x / 2;
    const midY = y + v_y / 2;
    const offset = 6; // 벡터와의 간격
    const textX = midX + offset * Math.cos(angle - Math.PI / 2);
    const textY = midY + offset * Math.sin(angle - Math.PI / 2);
    ctx.fillText(label, textX, textY);
    ctx.restore(); // 스타일 설정을 복구
  }

  draw();


</script>
