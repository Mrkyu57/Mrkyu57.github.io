<script>
  const canvas = document.getElementById('motionCanvas');
  const ctx = canvas.getContext('2d');

  const start = document.getElementById('start');
  const reset = document.getElementById('reset');
  const comBtn = document.getElementById('Com');
  const iniBtn = document.getElementById('initial');

  const radius = 15;                                              // 물체의 반지름
  const origin_position = { x: 450, y: 200 };                     // 원점의 위치
  
  let initial_position = { x: 450, y: 200 };                      // 처음 물체의 위치
  let initial_velocity = { x: 0.5, y: -0.1 };                     // 처음 물체의 속도
  let position = { x: 450, y: 200 };                              // 위치 벡터
  let velocity = { x: 0.5, y: -0.1 };                             // 속도 벡터

  let isMoving = false;                                           // 처음엔 정지
  let isReset = true;
  let showInitial = false;
  let isDraggingObject = false;
  let isDraggingVelocity = false;
  let showComponents = false;                                     // 벡터 성분 표시 (직각좌표)



  start.addEventListener('click', () => {                         // 정지 버튼을 클릭한경우 >
    isMoving = !isMoving;                                         // isMoving의 bool값을 반대로 변환
    isReset = false;

    if (isMoving) {                                               // 움직이는 경우 >
        start.innerHTML = '<span>&#10074;&#10074;</span>';        // ▮▮로 버튼의 모양 변환
        start.style.background = "#ffc107";                     // 버튼 배경 색상은 노란색
        start.style.color = "#000";                             // 버튼 글씨 색상은 검정색  
    } 
    else {                                                        // 정지한 경우 >
        start.innerHTML = '<span>&#9654;</span>';                 // ▶로 버튼의 모양 변환
        start.style.background = "#000000";                     // 버튼 배경 색상은 검정색
        start.style.color = "#fff";                             // 버튼 글씨 색상은 하양색 
    }
  });

  reset.addEventListener('click', () => {                         // 리셋 버튼을 클릭한경우 >
    isMoving = false;
    isReset = true;
    position = { ...initial_position };
    velocity = { ...initial_velocity };
    start.innerHTML = '<span>&#9654;</span>';
    start.style.background = "#000000";
    start.style.color = "#fff";
  });

  comBtn.addEventListener('click', () => {
    showComponents = !showComponents;
    comBtn.style.background = 
    showComponents ? "#8354ce" : "#555";
  });

  iniBtn.addEventListener('click', () => {
    showInitial = !showInitial;
    iniBtn.style.background = 
    showInitial ? "#8354ce" : "#555";
  });


  function draw() {                                                                                       // 그리기 함수 >

    ctx.clearRect(0, 0, canvas.width, canvas.height);                                                     //캔버스 초기화

    if (isMoving) {                                                                                     // 움직이는 경우 >
      position.x += velocity.x;                                                                         // x 위치 계산
      position.y += velocity.y;                                                                         // y 위치 계산

      
    }

    if (showInitial === true && isReset === false) { 
      drawMoveVector(initial_position.x, initial_position.y, 
      initial_velocity.x * 80, initial_velocity.y * 80, "#ff9696", "v₀");
      drawMoveVector(origin_position.x, origin_position.y, 
      initial_position.x - origin_position.x, initial_position.y - origin_position.y, "#d3d3d396", "r₀");
      ctx.fillStyle = "#d3d3d3";
      ctx.beginPath();
      ctx.arc(initial_position.x, initial_position.y, 10, 0, Math.PI * 2);
      ctx.fill();
    }

    const dispX = position.x - origin_position.x;                                                         // 원점으로 부터의 x방향 위치 차이
    const dispY = position.y - origin_position.y;                                                         // 원점으로 부터의 y방향 위치 차이
    drawMoveVector(origin_position.x, origin_position.y, dispX, dispY, "gray", "r");                      // 변위벡터 그리기 :
    drawMoveVector(position.x, position.y, velocity.x * 80, velocity.y * 80, "red","v");                  // 속도벡터 그리기 :
    
    if (showComponents) {                                                                                 // 벡터 성분을 표시할때 >
      ctx.setLineDash([5, 5]);                                                                            // 선 스타일 - 점선
      drawMoveVector(origin_position.x, origin_position.y, dispX, 0, "#555", "rx");                     // 변위벡터 x축 분해 그리기 :
      drawMoveVector(origin_position.x + dispX, origin_position.y, 0, dispY, "#555", "ry");             // 변위벡터 y축 분해 그리기 :
      drawMoveVector(position.x, position.y, velocity.x * 80, 0, "red", "vx");                            // 속도벡터 x축 분해 그리기 :
      drawMoveVector(position.x + velocity.x * 80, position.y, 0, velocity.y * 80, "red", "vy");          // 속도벡터 y축 분해 그리기 :
      ctx.setLineDash([]);                                                                                // 선 스타일 - 실선
    }

    ctx.fillStyle = "gray";                                                                   // 원점 그리기 >
    ctx.beginPath();
    ctx.arc(origin_position.x, origin_position.y, 3, 0, Math.PI * 2);
    ctx.fill();

    ctx.fillStyle = "#a2a2a2";                                                              // 물체 그리기 >
    ctx.beginPath();
    ctx.arc(position.x, position.y, 10, 0, Math.PI * 2);
    ctx.fill();
    ctx.lineWidth = 2; 
    ctx.strokeStyle = "#000000"; 
    ctx.stroke();            
    ctx.closePath();

    requestAnimationFrame(draw);
  }





 function drawMoveVector(x, y, v_x, v_y, color, label) {
    if (v_x === 0 && v_y === 0) return;
    
    ctx.save(); // 함수 시작 시 현재 상태 저장 (필수!)
    
    const headLength = 10;                
    const angle = Math.atan2(v_y, v_x);

    // 1. 벡터 선(몸통) 그리기
    ctx.strokeStyle = color;
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(x, y);
    ctx.lineTo(x + v_x, y + v_y);
    ctx.stroke();

    // 2. 화살표 머리 그리기
    ctx.beginPath();
    ctx.moveTo(x + v_x, y + v_y);
    ctx.lineTo(
        x + v_x - headLength * Math.cos(angle - Math.PI / 7), 
        y + v_y - headLength * Math.sin(angle - Math.PI / 7)
    );
    ctx.lineTo(
        x + v_x - headLength * Math.cos(angle + Math.PI / 7), 
        y + v_y - headLength * Math.sin(angle + Math.PI / 7)
    );
    ctx.closePath();
    ctx.fillStyle = color;
    ctx.fill();

    // 3. 텍스트 좌표 계산
    const midX = x + v_x / 2;
    const midY = y + v_y / 2;
    const offset = 12; // 글자와 벡터 사이 간격 살짝 넓힘
    const textX = midX + offset * Math.cos(angle - Math.PI / 2);
    const textY = midY + offset * Math.sin(angle - Math.PI / 2);

    // 4. 아래첨자 로직 적용하여 출력
    const labelStr = String(label);
    const mainText = labelStr.charAt(0);      // 'v' 또는 'r'
    const subText = labelStr.substring(1);   // 'x', 'y', '0' 등 나머지

    ctx.textAlign = "left"; // 첨자 위치 계산을 위해 왼쪽 정렬 고정
    ctx.textBaseline = "middle";

    ctx.strokeStyle = "#f9f9f9";
    ctx.lineWidth = 6;
    ctx.lineJoin = "round";

    ctx.font = "bold 18px 'Times New Roman', serif";
    ctx.strokeText(mainText, textX, textY);
    ctx.fillText(mainText, textX, textY);

    if (subText) {
        const mainWidth = ctx.measureText(mainText).width; // 'v'의 너비 측정
        ctx.font = "bold 12px 'Times New Roman', serif";  // 첨자는 작게
        
        // 메인 글자 바로 옆(mainWidth) + 살짝 아래(+5)
        const subX = textX + mainWidth;
        const subY = textY + 5; 
        
        ctx.strokeText(subText, subX, subY);
        ctx.fillText(subText, subX, subY);
    }

    ctx.restore(); // 저장했던 상태로 복구 (투명도, 색상 등 초기화)
 }





  function getMousePos(e) {
    const rect = canvas.getBoundingClientRect();
    return { x: e.clientX - rect.left, y: e.clientY - rect.top };
  }

  // [마우스 드래그 로직 핵심 수정]
  window.addEventListener('mousemove', (e) => {
    if (!isDraggingObject && !isDraggingVelocity) return;
    const mouse = getMousePos(e);

    if (isDraggingObject) {
        // 1. 물체의 현재 위치 업데이트
        position.x = mouse.x;
        position.y = mouse.y;
        // 2. 초기 위치(Reset 시 돌아갈 곳)를 현재 마우스 위치로 저장
        initial_position.x = mouse.x;
        initial_position.y = mouse.y;
        // ※ origin_position(원점)은 건드리지 않으므로 고정됩니다.
    } else if (isDraggingVelocity) {
        velocity.x = (mouse.x - position.x) / 80;
        velocity.y = (mouse.y - position.y) / 80;
        initial_velocity.x = velocity.x;
        initial_velocity.y = velocity.y;
    }
  });

  // (나머지 mousedown, mouseup, cursor 처리 리스너는 동일)
  canvas.addEventListener('mousedown', (e) => {
    if (!isReset) return; 
    const mouse = getMousePos(e);
    const distToObject = Math.hypot(mouse.x - position.x, mouse.y - position.y);
    const vEnd = { x: position.x + velocity.x * 80, y: position.y + velocity.y * 80 };
    const distToVelocity = Math.hypot(mouse.x - vEnd.x, mouse.y - vEnd.y);
    if (distToObject < 20) isDraggingObject = true;
    else if (distToVelocity < 20) isDraggingVelocity = true;
  });

  window.addEventListener('mouseup', () => {
    isDraggingObject = false;
    isDraggingVelocity = false;
  });

  canvas.addEventListener('mousemove', (e) => {
    const mouse = getMousePos(e);
    const distToObject = Math.hypot(mouse.x - position.x, mouse.y - position.y);
    const vEnd = { x: position.x + velocity.x * 80, y: position.y + velocity.y * 80 };
    const distToVelocity = Math.hypot(mouse.x - vEnd.x, mouse.y - vEnd.y);
    if (distToObject < 20 || distToVelocity < 20) canvas.style.cursor = 'grab';
    else canvas.style.cursor = 'default';
    if (isDraggingObject || isDraggingVelocity) canvas.style.cursor = 'grabbing';
  });

  draw();

</script>
